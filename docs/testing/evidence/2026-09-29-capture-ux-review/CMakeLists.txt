cmake_minimum_required(VERSION 3.24)
project(CaptureUxReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets Network Test)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_executable(review_probe probe.cpp)
target_compile_options(review_probe PRIVATE /utf-8)
target_compile_definitions(review_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(review_probe PRIVATE "${REPO}/native_capture/engine" "${REPO}/native_capture/platform/windows")
target_link_libraries(review_probe PRIVATE "${REPO}/native_capture/build/Release/capture_ui.lib" "${REPO}/native_capture/build/Release/capture_core.lib" Qt6::Widgets Qt6::Network Qt6::Test user32 gdi32 wtsapi32 dwmapi ole32 uiautomationcore)
