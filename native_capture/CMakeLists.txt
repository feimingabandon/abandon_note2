cmake_minimum_required(VERSION 3.24)
project(AbandonCapture VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
option(ABANDON_CAPTURE_BUILD_TESTS "Build explicitly selected Capture tests" OFF)
option(ABANDON_CAPTURE_BUILD_INPUT_DRIVER "Build the Capture input test driver" OFF)
set(ABANDON_CAPTURE_TEST_NAMES "" CACHE STRING "Explicit semicolon-separated Capture CTest names")
set(ABANDON_CAPTURE_RESULT_DIRECTORY "${CMAKE_BINARY_DIR}/test-results" CACHE PATH "Directory for this Capture test attempt")
set(_capture_known_tests capture_core capture_ui capture_ux capture_selection capture_pin_halo capture_polish capture_clipboard)
if(ABANDON_CAPTURE_BUILD_TESTS)
  if("${ABANDON_CAPTURE_TEST_NAMES}" STREQUAL "")
    message(FATAL_ERROR "ABANDON_CAPTURE_BUILD_TESTS requires explicit ABANDON_CAPTURE_TEST_NAMES")
  endif()
  set(_capture_unique_tests ${ABANDON_CAPTURE_TEST_NAMES})
  list(REMOVE_DUPLICATES _capture_unique_tests)
  list(LENGTH ABANDON_CAPTURE_TEST_NAMES _capture_test_count)
  list(LENGTH _capture_unique_tests _capture_unique_count)
  if(NOT _capture_test_count EQUAL _capture_unique_count)
    message(FATAL_ERROR "Duplicate Capture test names are not allowed")
  endif()
  foreach(name IN LISTS ABANDON_CAPTURE_TEST_NAMES)
    if(NOT name IN_LIST _capture_known_tests)
      message(FATAL_ERROR "Unknown Capture test name: ${name}")
    endif()
  endforeach()
elseif(NOT "${ABANDON_CAPTURE_TEST_NAMES}" STREQUAL "")
  message(FATAL_ERROR "ABANDON_CAPTURE_TEST_NAMES requires ABANDON_CAPTURE_BUILD_TESTS")
endif()
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets Network)
if(ABANDON_CAPTURE_BUILD_TESTS)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
endif()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_library(capture_host SHARED host/capture_host.cpp)
set_target_properties(capture_host PROPERTIES AUTOMOC OFF)
add_library(capture_core STATIC engine/ImageDocument.cpp engine/DesktopLayout.cpp engine/SelectionSnap.cpp platform/windows/DesktopCapture.cpp platform/windows/DesktopTargets.cpp)
target_link_libraries(capture_core PUBLIC Qt6::Widgets gdi32 user32 wtsapi32 dwmapi ole32 uiautomationcore)
target_include_directories(capture_core PUBLIC engine platform/windows)
add_library(capture_ui STATIC engine/CaptureEngine.cpp engine/CaptureOverlay.cpp engine/PinWindow.cpp engine/PinHalo.cpp engine/AnnotationEditor.cpp engine/CaptureAppearance.cpp engine/CaptureFeedback.cpp engine/SaveActions.cpp)
target_link_libraries(capture_ui PUBLIC capture_core Qt6::Network)
add_executable(AbandonCapture WIN32 engine/main.cpp)
target_link_libraries(AbandonCapture PRIVATE capture_ui)
enable_testing()
set(_capture_runtime_targets capture_host AbandonCapture)
if(ABANDON_CAPTURE_BUILD_TESTS)
  file(MAKE_DIRECTORY "${ABANDON_CAPTURE_RESULT_DIRECTORY}")
  foreach(name IN LISTS ABANDON_CAPTURE_TEST_NAMES)
    string(REGEX REPLACE "^capture_" "" suffix "${name}")
    set(target "${name}_tests")
    add_executable(${target} "tests/${suffix}_test.cpp")
    if(name STREQUAL "capture_core" OR name STREQUAL "capture_clipboard")
      target_link_libraries(${target} PRIVATE capture_core Qt6::Test)
    else()
      target_link_libraries(${target} PRIVATE capture_ui Qt6::Test)
    endif()
    add_test(NAME ${name} COMMAND ${CMAKE_COMMAND}
      "-DTEST_EXECUTABLE=$<TARGET_FILE:${target}>"
      "-DTEST_NAME=${name}"
      "-DDEFAULT_RESULT_DIRECTORY=${ABANDON_CAPTURE_RESULT_DIRECTORY}"
      -P "${CMAKE_SOURCE_DIR}/run-qt-test.cmake")
    set_tests_properties(${name} PROPERTIES TIMEOUT 90)
    if(name STREQUAL "capture_ux")
      # Popup focus, mouse capture and animation require the supported native platform.
      set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows")
    endif()
    list(APPEND _capture_runtime_targets ${target})
  endforeach()
endif()
if(ABANDON_CAPTURE_BUILD_INPUT_DRIVER)
  add_executable(capture_input_driver tests/input_driver.cpp)
  set_target_properties(capture_input_driver PROPERTIES AUTOMOC OFF)
  list(APPEND _capture_runtime_targets capture_input_driver)
endif()
foreach(target IN LISTS _capture_runtime_targets)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/bin")
endforeach()
