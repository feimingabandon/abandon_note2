cmake_minimum_required(VERSION 3.24)
project(AbandonCapture VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets Network Test)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
add_library(capture_host SHARED host/capture_host.cpp)
set_target_properties(capture_host PROPERTIES AUTOMOC OFF)
add_library(capture_core STATIC engine/ImageDocument.cpp engine/DesktopLayout.cpp engine/SelectionSnap.cpp platform/windows/DesktopCapture.cpp platform/windows/DesktopTargets.cpp)
target_link_libraries(capture_core PUBLIC Qt6::Widgets gdi32 user32 wtsapi32 dwmapi ole32 uiautomationcore)
target_include_directories(capture_core PUBLIC engine platform/windows)
add_library(capture_ui STATIC engine/CaptureEngine.cpp engine/CaptureOverlay.cpp engine/PinWindow.cpp engine/PinHalo.cpp engine/AnnotationEditor.cpp engine/CaptureAppearance.cpp engine/CaptureFeedback.cpp engine/SaveActions.cpp)
target_link_libraries(capture_ui PUBLIC capture_core Qt6::Network)
add_executable(AbandonCapture WIN32 engine/main.cpp)
target_link_libraries(AbandonCapture PRIVATE capture_ui)
enable_testing()
add_executable(capture_polish_tests tests/polish_test.cpp)
target_link_libraries(capture_polish_tests PRIVATE capture_ui Qt6::Test)
add_test(NAME capture_polish COMMAND capture_polish_tests -o "${CMAKE_BINARY_DIR}/capture-polish-results.txt,txt")
set_tests_properties(capture_polish PROPERTIES TIMEOUT 90)
add_executable(capture_selection_tests tests/selection_test.cpp)
target_link_libraries(capture_selection_tests PRIVATE capture_ui Qt6::Test)
add_test(NAME capture_selection COMMAND capture_selection_tests -o "${CMAKE_BINARY_DIR}/capture-selection-results.txt,txt")
set_tests_properties(capture_selection PROPERTIES TIMEOUT 90)
add_executable(capture_pin_halo_tests tests/pin_halo_test.cpp)
target_link_libraries(capture_pin_halo_tests PRIVATE capture_ui Qt6::Test)
add_test(NAME capture_pin_halo COMMAND capture_pin_halo_tests -o "${CMAKE_BINARY_DIR}/capture-pin-halo-results.txt,txt")
set_tests_properties(capture_pin_halo PROPERTIES TIMEOUT 90)
add_executable(capture_ux_tests tests/ux_test.cpp)
target_link_libraries(capture_ux_tests PRIVATE capture_ui Qt6::Test)
add_test(NAME capture_ux COMMAND capture_ux_tests -o "${CMAKE_BINARY_DIR}/capture-ux-results.txt,txt")
set_tests_properties(capture_ux PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(capture_core_tests tests/core_test.cpp)
target_link_libraries(capture_core_tests PRIVATE capture_core Qt6::Test)
add_test(NAME capture_core COMMAND capture_core_tests -o "${CMAKE_BINARY_DIR}/capture-core-results.txt,txt")
add_executable(capture_ui_tests tests/ui_test.cpp)
add_executable(capture_input_driver tests/input_driver.cpp)
set_target_properties(capture_input_driver PROPERTIES AUTOMOC OFF)
target_link_libraries(capture_ui_tests PRIVATE capture_ui Qt6::Test)
add_test(NAME capture_ui COMMAND capture_ui_tests -o "${CMAKE_BINARY_DIR}/capture-ui-results.txt,txt")
add_executable(capture_clipboard_tests tests/clipboard_test.cpp)
target_link_libraries(capture_clipboard_tests PRIVATE capture_core Qt6::Test)
add_test(NAME capture_clipboard COMMAND capture_clipboard_tests -o "${CMAKE_BINARY_DIR}/capture-clipboard-results.txt,txt")
set_tests_properties(capture_core capture_ui capture_clipboard PROPERTIES TIMEOUT 90)
foreach(target capture_host AbandonCapture capture_core_tests capture_ui_tests capture_ux_tests capture_selection_tests capture_pin_halo_tests capture_polish_tests capture_clipboard_tests capture_input_driver)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/bin")
endforeach()
